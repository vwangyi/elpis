## ts的typeof 

- typeof作用于类 得到的是 类的构造函数
- typeof 表示获取某个数据的类型
```ts
const a: string = 'safa';

let b: typeof a; // a是什么类型 b就是什么类型


```
```ts
class User {
  loginId: string
  loginPwd: string
}
function createUser(cls: new () => User): User {
  return new cls();
}
function createUser(cls: typeof User): User {
  return new cls();
}
const u = createUser(User);
```

## ts的keyof 
- keyof可以作用于类、接口、类型别名 等等。 用于获取其他类型中所有成员组成的联合类型
```ts
interface User {
  loginId: string
  loginPwd: string
  age: number
}
function printUser(obj: User, prop: 'loginId'|'loginPwd'|'age') {
   console.log(obj[prop])
}
function printUser(obj: User, prop: keyof User ) {
   console.log(obj[prop])
}
const u = User = {

  loginId: 'string'
  loginPwd: 'string'
  age: 34
}
printUser(u, 'age')
```

## in
- in关键字 经常和 keyof 连用，限制某个索引器的取值范围。

- 

```ts
interface User {
  loginId: string
  loginpwd: string
  age: number
}
type Obj = {
  [p: 'loginId' |"loginpwd"|'loginpwd']: string 
}
type Obj = {
  [p in 'loginId' |"loginpwd"|'loginpwd']: string  // 等同于下面
}
type Obj = {
  loginId: string
  loginpwd: string
  age: string
}

type Obj = {
  [p in keyof User]: string  
}

type Obj = {
  [p in keyof User]: User[p]  
}
type Obj = {
  readonly [p in keyof User]: User[p]  
}
type Obj = {
  [p in keyof User]?: User[p]  
}

type String<T> = {
  [p in keyof T]: string
}

type Readonly<T> = {
  readonly [p in keyof T]: T[p]
}

type Partial<T> = {
  [p in keyof T]?: T[p]
}
```
